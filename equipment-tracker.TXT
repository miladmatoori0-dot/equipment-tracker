<script>
const EQUIPMENT = [
  { key: 'case',      label: 'کیس' },
  { key: 'monitor',   label: 'مانیتور' },
  { key: 'printer',   label: 'پرینتر' },
  { key: 'scanner',   label: 'اسکنر' },
  { key: 'keyboard',  label: 'کیبورد' },
  { key: 'mouse',     label: 'ماوس' },
  { key: 'speaker',   label: 'اسپیکر' },
  { key: 'headphone', label: 'هدفون' },
  { key: 'other',     label: 'سایر', hasName: true }
];

const formsDiv = document.getElementById('equipmentForms');
EQUIPMENT.forEach(eq => {
  let html = `<div class="card"><h3>${eq.label}</h3>`;
  if (eq.hasName) {
    html += `<label>نام تجهیز</label><input type="text" id="other_name" placeholder="مثلاً وبکم">`;
  }
  html += `
    <label>شرکت سازنده</label>
    <input type="text" id="${eq.key}_brand" placeholder="مثلاً Dell">
    <label>مدل دستگاه</label>
    <input type="text" id="${eq.key}_model" placeholder="مثلاً OptiPlex 3080">
    <label>شماره اموال</label>
    <input type="text" id="${eq.key}_asset">
    <label>شماره سریال</label>
    <input type="text" id="${eq.key}_serial">
  </div>`;
  formsDiv.innerHTML += html;
});

function getDB() {
  return JSON.parse(localStorage.getItem('equipmentDB') || '{}');
}
function setDB(db) {
  localStorage.setItem('equipmentDB', JSON.stringify(db));
}

function savePerson() {
  const name = document.getElementById('personName').value.trim();
  if (!name) { showToast('نام فرد را وارد کنید', '#e74c3c'); return; }

  const data = {};
  EQUIPMENT.forEach(eq => {
    const item = {
      brand:  document.getElementById(eq.key + '_brand').value.trim(),
      model:  document.getElementById(eq.key + '_model').value.trim(),
      asset:  document.getElementById(eq.key + '_asset').value.trim(),
      serial: document.getElementById(eq.key + '_serial').value.trim()
    };
    if (eq.hasName) item.name = document.getElementById('other_name').value.trim();
    if (item.brand || item.model || item.asset || item.serial || (item.name && item.name !== '')) {
      data[eq.key] = item;
    }
  });

  const db = getDB();
  const wasExisting = db[name] !== undefined;
  db[name] = data;
  setDB(db);
  showToast(wasExisting ? '✅ ویرایش شد: ' + name : '✅ ذخیره شد: ' + name);
  clearForm();
}

function clearForm() {
  document.getElementById('personName').value = '';
  EQUIPMENT.forEach(eq => {
    ['_brand', '_model', '_asset', '_serial'].forEach(s => {
      document.getElementById(eq.key + s).value = '';
    });
    if (eq.hasName) document.getElementById('other_name').value = '';
  });
}

function showToast(msg, color = '#27ae60') {
  const t = document.getElementById('toast');
  t.textContent = msg;
  t.style.background = color;
  t.classList.add('show');
  setTimeout(() => t.classList.remove('show'), 2500);
}

function showSearch() {
  document.getElementById('mainPage').classList.add('hidden');
  document.getElementById('listPage').classList.add('hidden');
  document.getElementById('searchPage').classList.remove('hidden');
  document.getElementById('searchInput').value = '';
  document.getElementById('searchResults').innerHTML = '';
}

function doSearch() {
  const q = document.getElementById('searchInput').value.trim();
  const results = document.getElementById('searchResults');
  if (!q) { results.innerHTML = ''; return; }

  const db = getDB();
  const matches = Object.keys(db).filter(n => n.includes(q));

  if (matches.length === 0) {
    results.innerHTML = '<div class="empty">فردی یافت نشد</div>';
    return;
  }

  results.innerHTML = matches.map(name => {
    let html = `<div class="result-item"><b>👤 ${name}</b><br>`;
    EQUIPMENT.forEach(eq => {
      const item = db[name][eq.key];
      if (!item) return;
      const title = eq.hasName && item.name ? item.name : eq.label;
      const parts = [];
      if (item.brand)  parts.push('برند: ' + item.brand);
      if (item.model)  parts.push('مدل: ' + item.model);
      if (item.asset)  parts.push('اموال: ' + item.asset);
      if (item.serial) parts.push('سریال: ' + item.serial);
      if (parts.length) html += `<div style="margin:6px 0; font-size:13px;">🔹 <b>${title}</b><br>${parts.join(' | ')}</div>`;
    });
    html += '</div>';
    return html;
  }).join('');
}

function showSavedList() {
  document.getElementById('mainPage').classList.add('hidden');
  document.getElementById('searchPage').classList.add('hidden');
  document.getElementById('listPage').classList.remove('hidden');

  const db = getDB();
  const names = Object.keys(db).sort();
  const container = document.getElementById('savedList');

  if (names.length === 0) {
    container.innerHTML = '<div class="empty">هنوز فردی ثبت نشده</div>';
    return;
  }

  container.innerHTML = names.map(n =>
    `<div class="saved-person" onclick="viewPerson('${n.replace(/'/g, "\\'")}')">👤 ${n}</div>`
  ).join('');
}

function viewPerson(name) {
  showSearch();
  document.getElementById('searchInput').value = name;
  doSearch();
}

function goHome() {
  document.getElementById('searchPage').classList.add('hidden');
  document.getElementById('listPage').classList.add('hidden');
  document.getElementById('mainPage').classList.remove('hidden');
}

function backup() {
  const db = getDB();
  if (Object.keys(db).length === 0) {
    showToast('داده‌ای برای پشتیبان‌گیری نیست', '#e74c3c');
    return;
  }
  const blob = new Blob([JSON.stringify(db, null, 2)], { type: 'application/json' });
  const url = URL.createObjectURL(blob);
  const a = document.createElement('a');
  const date = new Date().toISOString().slice(0,10);
  a.href = url;
  a.download = `equipment-backup-${date}.json`;
  a.click();
  URL.revokeObjectURL(url);
  showToast('✅ فایل پشتیبان دانلود شد');
}

function restore(event) {
  const file = event.target.files[0];
  if (!file) return;
  const reader = new FileReader();
  reader.onload = (e) => {
    try {
      const data = JSON.parse(e.target.result);
      if (typeof data !== 'object' || Array.isArray(data)) throw new Error('bad');
      if (!confirm('همه داده‌های فعلی با فایل پشتیبان جایگزین می‌شوند. مطمئنی؟')) return;
      setDB(data);
      showToast('✅ بازیابی شد: ' + Object.keys(data).length + ' نفر');
      goHome();
    } catch (err) {
      showToast('فایل نامعتبر است', '#e74c3c');
    }
  };
  reader.readAsText(file);
  event.target.value = '';
}
